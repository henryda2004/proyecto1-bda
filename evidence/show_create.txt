===== stock =====

table_name	create_statement
stock	"CREATE TABLE public.stock (
	region public.crdb_internal_region NOT NULL,
	codproducto UUID NOT NULL,
	unidades INT8 NOT NULL,
	CONSTRAINT stock_pkey PRIMARY KEY (region ASC, codproducto ASC),
	CONSTRAINT stock_codproducto_fkey FOREIGN KEY (codproducto) REFERENCES public.producto(codproducto),
	CONSTRAINT check_unidades CHECK (unidades >= 0:::INT8)
) LOCALITY REGIONAL BY ROW AS region"

===== pedido =====

table_name	create_statement
pedido	"CREATE TABLE public.pedido (
	region public.crdb_internal_region NOT NULL,
	codpedido UUID NOT NULL DEFAULT gen_random_uuid(),
	codproducto UUID NOT NULL,
	codcliente STRING NOT NULL,
	unidades INT8 NOT NULL,
	fecha TIMESTAMPTZ NOT NULL DEFAULT now():::TIMESTAMPTZ,
	CONSTRAINT pedido_pkey PRIMARY KEY (region ASC, codpedido ASC),
	CONSTRAINT pedido_codproducto_fkey FOREIGN KEY (codproducto) REFERENCES public.producto(codproducto),
	CONSTRAINT check_unidades CHECK (unidades > 0:::INT8)
) LOCALITY REGIONAL BY ROW AS region"
